  <h2 id="architecture" className="smb-h2">Architecture</h2>

  <p className="smb-p">KRONOS couples a <strong>GAT graph encoder</strong> over the STRING PPI network with a <strong>Vicuna-7b-v1.5</strong> language backbone. Per-patient mass spectrometry abundances become node features (~4,800 proteins). The graph embedding is projected as a soft prefix to the LLM, which then generates predictions from natural language instructions.</p>

```python
import torch, numpy as np
from torch_geometric.data import Data
from transformers import AutoModelForCausalLM, AutoTokenizer

device = "cuda" if torch.cuda.is_available() else "cpu"
model = AutoModelForCausalLM.from_pretrained(
    "standardmodelbio/kronos", trust_remote_code=True,
    torch_dtype=torch.float16, device_map="auto",
).eval()
tokenizer = AutoTokenizer.from_pretrained(
    "standardmodelbio/kronos", trust_remote_code=True,
)

# Build patient graph
abundances = np.load("/path/to/patient_proteomics.npy")
node_features = torch.tensor(abundances, dtype=torch.float32)
edge_index = torch.load("/path/to/string_ppi_edge_index.pt")

patient_graph = Data(
    x=node_features.unsqueeze(0),
    edge_index=edge_index,
).to(device)

# Inference
instruction = "What is the predicted cancer type?"
inputs = tokenizer(instruction, return_tensors="pt").to(device)

with torch.no_grad():
    out = model.generate(
        **inputs, graph_data=patient_graph,
        max_new_tokens=64, do_sample=False,
    )

answer = tokenizer.decode(out[0], skip_special_tokens=True)
print(f"Prediction: {answer}")

# Extract graph embedding for downstream use
with torch.no_grad():
    graph_emb = model.encode_graph(patient_graph)
print(f"Patient embedding: {graph_emb.shape}")
```
