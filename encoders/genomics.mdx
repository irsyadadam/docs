```python
import torch, pandas as pd
from transformers import AutoModel, AutoTokenizer

device = "cuda" if torch.cuda.is_available() else "cpu"
model = AutoModel.from_pretrained(
    "standardmodelbio/genvarformer", trust_remote_code=True,
).to(device).eval()

# Mutation features for one gene (16 Mbp window)
mutations = pd.DataFrame({
    "ALT":     ["T", "GCTA", "A"],
    "ILEN":    [0, 3, -1],
    "VAF":     [0.42, 0.15, 0.87],
    "FLANK_5": ["ACGTACGT" * 4] * 3,
    "FLANK_3": ["GCTAGCTA" * 4] * 3,
    "POS":     [12345678, 12400000, 12800000],
})

tokenizer = AutoTokenizer.from_pretrained(
    "standardmodelbio/genvarformer", trust_remote_code=True,
)
inputs = tokenizer(
    mutations.to_dict("records"), return_tensors="pt"
).to(device)

with torch.no_grad():
    outputs = model(**inputs)

expr_pred = outputs.expression_logits        # scalar
gene_emb = outputs.last_hidden_state.mean(1) # (1, 64)

print(f"Expression delta: {expr_pred.item():.4f}")
print(f"Gene embedding:   {gene_emb.shape}")

# Patient-level: repeat for all genes, concatenate
```
